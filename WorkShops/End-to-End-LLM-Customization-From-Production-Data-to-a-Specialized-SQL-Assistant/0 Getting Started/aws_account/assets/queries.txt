WITH monthly AS (SELECT TO_CHAR(sale_date, 'YYYY-MM') as month, SUM(price * quantity) as revenue FROM product_sales GROUP BY month) SELECT month, revenue, ROUND(AVG(revenue) OVER (ORDER BY month ROWS BETWEEN 2 PRECEDING AND CURRENT ROW), 2) as moving_avg_3m FROM monthly ORDER BY month;

WITH luxury AS (SELECT country_code, SUM(price * quantity) as luxury_revenue FROM product_sales WHERE brand_tier = 'luxury' GROUP BY country_code), total AS (SELECT country_code, SUM(price * quantity) as total_revenue FROM product_sales GROUP BY country_code) SELECT t.country_code, l.luxury_revenue, t.total_revenue, ROUND(l.luxury_revenue * 100.0 / t.total_revenue, 2) as luxury_share_pct FROM total t JOIN luxury l ON t.country_code = l.country_code ORDER BY luxury_share_pct DESC;
WITH stats AS (SELECT AVG(price) as avg_price, STDDEV(price) as sd_price FROM product_sales) SELECT p.product_name, p.brand, p.price FROM product_sales p, stats s WHERE p.price > s.avg_price + 2 * s.sd_price ORDER BY p.price DESC;
WITH category_avg AS (SELECT product_category, AVG(rating) as cat_avg_rating FROM product_sales GROUP BY product_category) SELECT p.product_name, p.brand, p.product_category, p.rating, ROUND(c.cat_avg_rating, 2) as category_avg_rating FROM product_sales p JOIN category_avg c ON p.product_category = c.product_category WHERE p.rating > c.cat_avg_rating ORDER BY p.rating DESC LIMIT 25;
WITH yearly AS (SELECT EXTRACT(YEAR FROM sale_date) as year, product_category, SUM(price * quantity) as revenue FROM product_sales GROUP BY year, product_category) SELECT a.product_category, a.year as year_1, a.revenue as revenue_1, b.year as year_2, b.revenue as revenue_2, ROUND((b.revenue - a.revenue) * 100.0 / a.revenue, 2) as growth_pct FROM yearly a JOIN yearly b ON a.product_category = b.product_category AND b.year = a.year + 1 ORDER BY a.product_category, a.year;
